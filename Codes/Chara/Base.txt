// Example engine hook:
function renderEngineSprite(ctx, frameData, cellSize) {
    // Layer 1 (base)
    frameData.layer1.forEach((row, r) => {
        row.forEach((color, c) => {
            ctx.fillStyle = color;
            ctx.fillRect(c * cellSize, r * cellSize, cellSize, cellSize);
        });
    });
    // Layer 2 (overlay)
    frameData.layer2.forEach((row, r) => {
        row.forEach((color, c) => {
            if (color !== 'transparent') {
                ctx.fillStyle = color;
                ctx.fillRect(c * cellSize, r * cellSize, cellSize, cellSize);
            }
        });
    });
}
